# The asm! calls of the seams verus/isla/seams.md lists, verbatim up to
# whitespace: FILE<TAB>FN<TAB>ASM. `sh verus/seams.sh --check` fails when
# the asm! calls of a seam's item and these lines differ, so a changed
# template fails until its jobs are revisited.
src/sys/raw.rs	syscall0	asm!("svc 0", in("x8") nr, lateout("x0") ret, options(nostack));
src/sys/raw.rs	syscall1	asm!("svc 0", in("x8") nr, inlateout("x0") a0 => ret, options(nostack));
src/sys/raw.rs	syscall2	asm!("svc 0", in("x8") nr, inlateout("x0") a0 => ret, in("x1") a1, options(nostack));
src/sys/raw.rs	syscall3	asm!("svc 0", in("x8") nr, inlateout("x0") a0 => ret, in("x1") a1, in("x2") a2, options(nostack));
src/sys/raw.rs	syscall4	asm!("svc 0", in("x8") nr, inlateout("x0") a0 => ret, in("x1") a1, in("x2") a2, in("x3") a3, options(nostack));
src/sys/raw.rs	syscall5	asm!("svc 0", in("x8") nr, inlateout("x0") a0 => ret, in("x1") a1, in("x2") a2, in("x3") a3, in("x4") a4, options(nostack));
src/sys/raw.rs	syscall6	asm!("svc 0", in("x8") nr, inlateout("x0") a0 => ret, in("x1") a1, in("x2") a2, in("x3") a3, in("x4") a4, in("x5") a5, options(nostack));
src/sys/tls.rs	get_tp	asm!( "adrp {off}, :gottprel:challoc_tls", "ldr {off}, [{off}, :gottprel_lo12:challoc_tls]", "mrs {tp}, tpidr_el0", "ldr {out}, [{tp}, {off}]", off = out(reg) _, tp = out(reg) tp, out = out(reg) out, options(nostack, readonly, preserves_flags), );
