axiom fn axiom_heap_block(h: *mut Heap, heap: Heap, tracked perms: &mut CachePerms) -> (tracked r: (vstd::raw_ptr::PointsToRaw, crate::meta::MetaClaim))

        heap.wf(),
        heap.home_at(h),
        *old(perms) == heap.perms@,
    ensures
