axiom fn axiom_hand_out(p: *mut u8, len: int) -> (tracked m: UserMem)
    requires
        crate::heap::regions_ok(),
        process_regions().is_huge_block(p@.addr, len),
    ensures
        m.covers(p@.addr as int, len),
        m.provenance() == p@.provenance,
;
